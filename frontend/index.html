How Web APIs & FastAPI Work (The Restaurant Analogy)

If you've never built a web server before, think of FastAPI as a restaurant service pipeline:

1. Customer Frontend / Browser

Sends an order (HTTP Request) asking for data or creating a new item.

2. Menu & Rules Pydantic Schemas

Defines valid items and ensures the order parameters are correctly formatted.

3. Waiter FastAPI Router

Receives the order at an endpoint (e.g. /api/items) and passes it to the kitchen.

4. Kitchen & Chef Python Function

Executes business logic, checks credentials, and prepares the requested payload.

5. Pantry SQLAlchemy & Database

Where raw ingredients (database rows & tables) are fetched or stored safely.

6. Served Dish JSON Response

The final structured data sent back to the customer with an HTTP 200 OK status code.

FastAPI Backend Code Blueprint

Production Python main.py template connecting FastAPI, Pydantic, and SQLAlchemy.

# main.py - Clean FastAPI Backend with SQLAlchemy & Pydantic
from fastapi import FastAPI, Depends, HTTPException, status
from fastapi.middleware.cors import CORSMiddleware
from sqlalchemy.orm import Session
from database import Base, engine, SessionLocal, DB_ENGINE_TYPE
from database_model import ItemDB
from models import ItemCreate, ItemResponse

# Create database tables automatically
Base.metadata.create_all(bind=engine)

app = FastAPI(title="FastAPI Hub Backend", version="2.0.0")

app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

def get_db():
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()

@app.get("/health")
def health_check():
    return {"status": "ok", "engine": DB_ENGINE_TYPE}

@app.get("/api/v1/items", response_model=list[ItemResponse])
def get_all_items(db: Session = Depends(get_db)):
    return db.query(ItemDB).order_by(ItemDB.id.asc()).all()

@app.post("/api/v1/items", response_model=ItemResponse, status_code=status.HTTP_201_CREATED)
def create_item(item: ItemCreate, db: Session = Depends(get_db)):
    db_item = ItemDB(**item.model_dump())
    db.add(db_item)
    db.commit()
    db.refresh(db_item)
    return db_item

@app.put("/api/v1/items/{id}", response_model=ItemResponse)
def update_item(id: int, item: ItemCreate, db: Session = Depends(get_db)):
    db_item = db.query(ItemDB).filter(ItemDB.id == id).first()
    if not db_item:
        raise HTTPException(status_code=404, detail="Item not found")
    for key, val in item.model_dump().items():
        setattr(db_item, key, val)
    db.commit()
    db.refresh(db_item)
    return db_item

@app.delete("/api/v1/items/{id}")
def delete_item(id: int, db: Session = Depends(get_db)):
    db_item = db.query(ItemDB).filter(ItemDB.id == id).first()
    if not db_item:
        raise HTTPException(status_code=404, detail="Item not found")
    db.delete(db_item)
    db.commit()
    return {"detail": f"Item #{id} deleted"}

Pydantic Request & Response Validation

Define request and response schemas to validate JSON payloads automatically at runtime.

from pydantic import BaseModel

class ItemCreate(BaseModel):
    title: str
    category: str = "General"

class ItemResponse(ItemCreate):
    id: int
    completed: bool

SQLAlchemy Relational Models

Map Python classes to relational database tables using DeclarativeBase.

from sqlalchemy.orm import DeclarativeBase, Mapped, mapped_column

class Base(DeclarativeBase):
    pass

class ItemDB(Base):
    __tablename__ = "items"
    id: Mapped[int] = mapped_column(primary_key=True)
    title: Mapped[str]
